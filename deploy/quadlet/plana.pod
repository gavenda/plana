[Unit]
Description=Plana pod (bot and dashboard)
# Starts once the host's Valkey answers, or after the wait gives up (see ../valkey-wait.service).
Wants=valkey-wait.service
After=valkey-wait.service

[Pod]
PodName=plana

# Published here, not on the container: the pod owns the network namespace.
#
# Loopback on purpose: terminate TLS with a reverse proxy in front, and set
# PUBLIC_BASE_URL to the public https:// origin.
PublishPort=127.0.0.1:3000:3000

# Redis (Valkey) runs on the host, shared with Vivy, and listens on the host's loopback
# only. Inside this pod 127.0.0.1 is the pod itself, so pasta's -T
# forwards the pod's own port 6379 to the host's.
Network=pasta:-T,6379

[Install]
WantedBy=default.target
